lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  (m (k + 1) = (2 * m k + 1) / d) ∧ (n (k + 1) = (2 * n k + 1) / d) := by
  -- obtain the successor coprimality hypothesis for index `k+1`
  have hsucc_copr : Nat.Coprime (m (k + 1)) (n (k + 1)) := by
    exact h_coprime (k + 1) (Nat.succ_pos k)

  -- the right-hand denominator (2 * n k + 1) is a nonzero natural number
  have denom_nonzero : (2 * n k + 1) ≠ 0 := by
    simp [Nat.succ_ne_zero]

  -- apply the reduced-fraction uniqueness lemma to the equality of rationals
  have hpair : (m (k + 1) = (2 * m k + 1) / Nat.gcd (2 * m k + 1) (2 * n k + 1)) ∧
               (n (k + 1) = (2 * n k + 1) / Nat.gcd (2 * m k + 1) (2 * n k + 1)) := by
    exact reduced_fraction_unique (m (k + 1)) (n (k + 1)) (2 * m k + 1) (2 * n k + 1)
      hsucc_copr denom_nonzero (h_recurrence k)

  exact hpair
