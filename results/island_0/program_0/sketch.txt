lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  (m (k + 1) = (2 * m k + 1) / d) ∧ (n (k + 1) = (2 * n k + 1) / d) := by
  sorry

/-- Step 2: the differences never become zero. -/
lemma diff_ne_zero
  (m n : ℕ → ℕ)
  (h_distinct : m 0 ≠ n 0)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
  sorry

/-- Step 3: `d_k = gcd(2m_k+1,2n_k+1)` is odd and divides the difference. -/
lemma gcd_odd_and_dvd_diff
  (m n : ℕ → ℕ)
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  d % 2 = 1 ∧ ((d : ℤ) ∣ ((m k : ℤ) - (n k : ℤ))) := by
  sorry

/-- Step 4: define the "odd part" measure from the absolute difference. -/
lemma oddPart_measure_pos
  (m n : ℕ → ℕ)
  (hΔ : ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0) :
  ∀ k : ℕ,
    0 < Int.natAbs ((m k : ℤ) - (n k : ℤ)) := by
  sorry

/-- Step 4/5: if `d_k > 1`, the odd-part measure strictly decreases. -/
lemma oddPart_strict_decrease_of_not_coprime
  (m n : ℕ → ℕ)
  (hΔ : ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1))
  (k : ℕ)
  (hbad : ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)) :
  Int.natAbs ((m (k + 1) : ℤ) - (n (k + 1) : ℤ)) <
    Int.natAbs ((m k : ℤ) - (n k : ℤ)) := by
  sorry

/-- Step 5: if a positive-ℕ valued measure strictly decreases whenever `bad k`, then `{k | bad k}` is finite. -/
lemma finite_of_strict_decrease_on_bad
  (t : ℕ → ℕ)
  (htpos : ∀ k : ℕ, 0 < t k)
  (bad : ℕ → Prop)
  (hdec : ∀ k : ℕ, bad k → t (k + 1) < t k) :
  {k : ℕ | bad k}.Finite := by
  sorry

theorem putnam_2025_a1
  (m n : ℕ → ℕ)
  (hm : ∀ k : ℕ, 0 < m k)
  (hn : ∀ k : ℕ, 0 < n k)
  (h_distinct : m 0 ≠ n 0)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  {k : ℕ | ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)}.Finite := by
  classical

  -- Step 2: differences never vanish
  have hΔ : ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
    exact diff_ne_zero m n h_distinct h_recurrence

  -- Step 4: define the decreasing measure
  let t : ℕ → ℕ := fun k =>
    Int.natAbs ((m k : ℤ) - (n k : ℤ))

  have htpos : ∀ k : ℕ, 0 < t k := by
    intro k
    -- unfold `t` via the lemma
    simpa [t] using (oddPart_measure_pos m n hΔ k)

  -- Step 4/5: whenever the gcd is nontrivial (i.e. not coprime), the measure decreases
  let bad : ℕ → Prop := fun k => ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)

  have hdec : ∀ k : ℕ, bad k → t (k + 1) < t k := by
    intro k hbad
    -- direct invocation of the decrease lemma
    simpa [t, bad] using
      (oddPart_strict_decrease_of_not_coprime m n hΔ h_coprime h_recurrence k hbad)

  -- Step 5: apply the abstract finiteness lemma
  have hfinite : {k : ℕ | bad k}.Finite := by
    exact finite_of_strict_decrease_on_bad t htpos bad hdec

  -- conclude in the required form
  simpa [bad] using hfinite
