/-- Helper: lift the `h_coprime` hypothesis to the successor index. -/
lemma helper_succ_coprime
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (k : ℕ) :
  Nat.Coprime (m (k + 1)) (n (k + 1)) := by
  -- uses h_coprime at index `k+1` and `0 < k+1`
  exact h_coprime (k + 1) (Nat.zero_lt_succ k)

lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  (m (k + 1) = (2 * m k + 1) / d) ∧ (n (k + 1) = (2 * n k + 1) / d) := by
  -- obtain nonvanishing facts for denominators from the recurrence helper already present
  have hden : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 ∧ ((2 : ℚ) * (m k : ℚ) + 1) ≠ 0 := by
    exact denom_nonzero_from_recurrence m n h_recurrence

  -- get that the successor denominator `(n (k+1) : ℚ)` is nonzero
  have hsucc_nz : (n (k + 1) : ℚ) ≠ 0 := by
    exact n_succ_nonzero m n h_recurrence k

  -- successor coprimality (from the input hypothesis)
  have hcop_succ : Nat.Coprime (m (k + 1)) (n (k + 1)) := by
    exact helper_succ_coprime m n h_coprime k

  -- apply the reduced-form uniqueness lemma to deduce the gcd-quotient equalities
  have hres : let d := Nat.gcd (2 * m k + 1) (2 * n k + 1) in
    (m (k + 1) = (2 * m k + 1) / d) ∧ (n (k + 1) = (2 * n k + 1) / d) := by
    exact helper_reduced_form_uniqueness (m (k + 1)) (n (k + 1)) (2 * m k + 1) (2 * n k + 1)
      (by exact hsucc_nz) (by exact (hden k).1) hcop_succ (h_recurrence k)

  exact hres
