lemma finite_strict_drops_of_bound (t : ℕ → ℕ) (N : ℕ) (h0 : t 0 ≤ N)
  (hnoninc : ∀ k, t (k + 1) ≤ t k) :
  {k : ℕ | t (k + 1) < t k}.Finite := by
  -- obtain the global bound t k ≤ N
  have hglob : ∀ k, t k ≤ N := by
    exact global_bound_from_h0 t N h0 hnoninc

  -- bound the cardinality of the set of drops below any K
  have hcard_bound : ∀ K, (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ N := by
    exact fin_card_bound t N h0 hnoninc

  -- apply the abstract conversion from uniform cardinal bounds to finiteness
  have hfin : {k : ℕ | t (k + 1) < t k}.Finite := by
    exact finite_of_card_bounded N hcard_bound
