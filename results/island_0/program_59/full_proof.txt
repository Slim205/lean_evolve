theorem nat_cast_nonneg (a : ℕ) : (a : ℚ) ≥ 0 := by
  exact_mod_cast Nat.zero_le a
theorem two_nonneg : (2 : ℚ) ≥ 0 := by norm_num
theorem mul_nonneg_of_nonneg {x y : ℚ} (hx : x ≥ 0) (hy : y ≥ 0) : x * y ≥ 0 := by
  apply mul_nonneg hx hy
theorem one_pos_q : (1 : ℚ) > 0 := by norm_num
theorem add_pos_of_nonneg_and_pos {x y : ℚ} (hx : x ≥ 0) (hy : y > 0) : x + y > 0 := by
  have : x + y > 0 := by linarith
  exact this
theorem pos_ne_zero {x : ℚ} (h : x > 0) : x ≠ 0 := by
  exact ne_of_gt h
lemma two_mul_nat_plus_one_nonzero (a : ℕ) : ((2 : ℚ) * (a : ℚ) + 1) ≠ 0 := by
  have ha_nonneg : (a : ℚ) ≥ 0 := by exact nat_cast_nonneg a
  have h2_nonneg : (2 : ℚ) ≥ 0 := by exact two_nonneg
  have hmul_nonneg : (2 : ℚ) * (a : ℚ) ≥ 0 := by exact mul_nonneg_of_nonneg h2_nonneg ha_nonneg
  have hone_pos : (1 : ℚ) > 0 := by exact one_pos_q
  have hsum_pos : (2 : ℚ) * (a : ℚ) + 1 > 0 := by exact add_pos_of_nonneg_and_pos hmul_nonneg hone_pos
  exact pos_ne_zero hsum_pos

theorem recurrence_cross_Q (m n : ℕ → ℕ) (k : ℕ)
  (h_recurrence : ∀ j : ℕ,
    (m (j + 1) : ℚ) / (n (j + 1) : ℚ) =
      ((2 : ℚ) * (m j : ℚ) + 1) / ((2 : ℚ) * (n j : ℚ) + 1))
  (h_n_succ_nonzero_k : (n (k + 1) : ℚ) ≠ 0)
  (h_two_nk_nonzero : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0) :
  (m (k + 1) : ℚ) * ((2 : ℚ) * (n k : ℚ) + 1) = (n (k + 1) : ℚ) * ((2 : ℚ) * (m k : ℚ) + 1) := by
  have h1 : (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1) := by
    apply h_recurrence
  have h2 : (n (k + 1) : ℚ) ≠ 0 := by exact_mod_cast h_n_succ_nonzero_k
  have h3 : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 := by exact_mod_cast h_two_nk_nonzero
  field_simp at h1
  nlinarith
theorem nat_cast_inj_mul {a b c d : ℕ}
  (h : ((a : ℚ) * (b : ℚ)) = ((c : ℚ) * (d : ℚ))) : a * b = c * d := by
  ring_nf at h
  norm_cast at h ⊢
  <;>
  simp_all [mul_comm, mul_assoc, mul_left_comm]
  <;>
  omega
lemma cross_mul_eq_nat
  (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) =
      ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (k : ℕ)
  (h_n_succ_nonzero_k : (n (k + 1) : ℚ) ≠ 0)
  (h_two_nk_nonzero : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0) :
  m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
  have hq : (m (k + 1) : ℚ) * ((2 : ℚ) * (n k : ℚ) + 1) =
            (n (k + 1) : ℚ) * ((2 : ℚ) * (m k : ℚ) + 1) := by
    exact recurrence_cross_Q m n k h_recurrence h_n_succ_nonzero_k h_two_nk_nonzero

  have hleft_cast : ((2 : ℚ) * (n k : ℚ) + 1) = ((2 * n k + 1 : ℕ) : ℚ) := by
    exact cast_two_mul_nat_plus_one (n k)
  have hright_cast : ((2 : ℚ) * (m k : ℚ) + 1) = ((2 * m k + 1 : ℕ) : ℚ) := by
    exact cast_two_mul_nat_plus_one (m k)

  have hq_natcasts : (m (k + 1) : ℚ) * ((2 * n k + 1 : ℕ) : ℚ) =
                     (n (k + 1) : ℚ) * ((2 * m k + 1 : ℕ) : ℚ) := by
    rw [hleft_cast, hright_cast] at hq
    exact hq

  have hnat : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
    apply nat_cast_inj_mul
    exact hq_natcasts
  exact hnat

theorem build_common_scaling_from_divs (m n : ℕ → ℕ) (k : ℕ)
  (cross_eq : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1))
  (hm_div : m (k + 1) ∣ (2 * m k + 1))
  (hn_div : n (k + 1) ∣ (2 * n k + 1)) :
  ∃ c : ℕ, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
  have h₁ : m (k + 1) ∣ (2 * m k + 1) := hm_div
  have h₂ : n (k + 1) ∣ (2 * n k + 1) := hn_div
  have h₃ : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := cross_eq
  have h₄ : ∃ c : ℕ, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
    have h₅ : m (k + 1) ∣ (2 * m k + 1) := hm_div
    have h₆ : n (k + 1) ∣ (2 * n k + 1) := hn_div
    have h₇ : m (k + 1) ∣ (2 * m k + 1) := hm_div
    have h₈ : n (k + 1) ∣ (2 * n k + 1) := hn_div
    obtain ⟨d, hd⟩ := h₅
    obtain ⟨e, he⟩ := h₆
    use d
    constructor
    · rw [hd]
      nlinarith
    · rw [he]
      nlinarith
  exact h₄
lemma next_scaling_exists
  (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) =
      ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (h_n_succ_nonzero : ∀ k, (n (k + 1) : ℚ) ≠ 0)
  (h_coprime_succ : ∀ k, Nat.Coprime (m (k + 1)) (n (k + 1)))
  (k : ℕ) :
  ∃ c : ℕ, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
  have h2n_nonzero : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 := by
    exact two_mul_nat_plus_one_nonzero (n k)

  have hcross : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
    exact cross_mul_eq_nat m n h_recurrence k (h_n_succ_nonzero k) h2n_nonzero

  have hm_div_mul : m (k + 1) ∣ n (k + 1) * (2 * m k + 1) := by
    have : n (k + 1) * (2 * m k + 1) = m (k + 1) * (2 * n k + 1) := by
      exact Eq.symm hcross
    rw [this]
    exact dvd_mul_right _ _

  have hm_div : m (k + 1) ∣ (2 * m k + 1) := by
    exact @coprime_dvd_of_dvd_mul_right (m (k + 1)) (n (k + 1)) (2 * m k + 1) (h_coprime_succ k) hm_div_mul

  have hn_div_mul : n (k + 1) ∣ m (k + 1) * (2 * n k + 1) := by
    have : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
      exact hcross
    rw [this]
    exact dvd_mul_right _ _
  have hn_div : n (k + 1) ∣ (2 * n k + 1) := by
    exact @coprime_dvd_of_dvd_mul_right (n (k + 1)) (m (k + 1)) (2 * n k + 1) (Nat.Coprime.symm (h_coprime_succ k)) hn_div_mul

  have hex : ∃ c, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
    exact build_common_scaling_from_divs m n k hcross hm_div hn_div
  exact hex
theorem scaling_factor_eq_gcd (m n : ℕ → ℕ) (k c : ℕ)
  (hc1 : (2 * m k + 1) = c * m (k + 1))
  (hc2 : (2 * n k + 1) = c * n (k + 1))
  (h_coprime_succ : Nat.Coprime (m (k + 1)) (n (k + 1))) :
  c = Nat.gcd (2 * m k + 1) (2 * n k + 1) := by
  have h₁ : c = Nat.gcd (2 * m k + 1) (2 * n k + 1) := by
    rw [show (2 * m k + 1) = c * m (k + 1) by omega]
    rw [show (2 * n k + 1) = c * n (k + 1) by omega]
    rw [Nat.gcd_mul_left]
    <;> simp [h_coprime_succ, Nat.gcd_eq_right]
  exact h₁
theorem n_succ_nonzero (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  ∀ k, (n (k + 1) : ℚ) ≠ 0 := by
  intro k
  have h1 : (n (k + 1) : ℚ) ≠ 0 := by
    have h2 : (n (k + 1) : ℚ) ≠ 0 := by
      intro h
      have h3 := h_recurrence k
      field_simp [h] at h3
      <;> norm_cast at h3 ⊢ <;> ring_nf at h3 ⊢ <;> omega
    exact h2
  exact h1
lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) =
      ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  ((2 * m k + 1) = d * m (k + 1)) ∧ ((2 * n k + 1) = d * n (k + 1)) := by
  have hscal : ∃ c, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by

    exact next_scaling_exists m n h_recurrence (n_succ_nonzero m n h_recurrence)
      (fun i => h_coprime (i + 1) (Nat.succ_pos i)) k
  rcases hscal with ⟨c, ⟨hc1, hc2⟩⟩

  have hc_gcd : c = Nat.gcd (2 * m k + 1) (2 * n k + 1) := by
    have succ_copr : Nat.Coprime (m (k + 1)) (n (k + 1)) := h_coprime (k + 1) (Nat.succ_pos k)
    exact scaling_factor_eq_gcd m n k c hc1 hc2 succ_copr

  let d := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  have : (2 * m k + 1) = d * m (k + 1) := by
    rw [hc_gcd] at hc1
    exact hc1
  have : (2 * n k + 1) = d * n (k + 1) := by
    rw [hc_gcd] at hc2
    exact hc2
  exact ⟨by assumption, by assumption⟩
theorem denom_nonzero_from_recurrence (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  ∀ k, ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 ∧ ((2 : ℚ) * (m k : ℚ) + 1) ≠ 0 := by
  intro k
  have h₁ : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 ∧ ((2 : ℚ) * (m k : ℚ) + 1) ≠ 0 := by
    have h₂ := h_recurrence k
    have h₃ := h_recurrence 0
    have h₄ := h_recurrence 1
    have h₅ := h_recurrence 2
    norm_num at h₂ h₃ h₄ h₅
    <;>
    (try simp_all [div_eq_mul_inv])
    <;>
    (try norm_num)
    <;>
    (try field_simp at * )
    <;>
    (try ring_nf at * )
    <;>
    (try norm_cast at * )
    <;>
    (try omega)
  exact h₁
theorem succ_equal_implies_ratio_one (m n : ℕ → ℕ)
  (h_n_succ_nonzero : ∀ k, (n (k + 1) : ℚ) ≠ 0)
  (k : ℕ) :
  m (k + 1) = n (k + 1) →
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = 1 := by
  intro h
  have h₁ : (n (k + 1) : ℚ) ≠ 0 := h_n_succ_nonzero k
  field_simp [h₁] at h ⊢
  <;> simp_all
theorem ratio_one_implies_eq_prev (m n : ℕ → ℕ)
  (h_denom_nonzero : ∀ k, ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0)
  (k : ℕ) :
  (((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1) = 1) → m k = n k := by
  intro h
  have h1 : ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1) = 1 := h
  have h2 : (2 : ℚ) * (n k : ℚ) + 1 ≠ 0 := h_denom_nonzero k
  field_simp at h1
  ring_nf at h1
  norm_cast at h1
  <;> omega

lemma diff_ne_zero
  (m n : ℕ → ℕ)
  (h_distinct : m 0 ≠ n 0)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
  have hden : ∀ k, ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 ∧ ((2 : ℚ) * (m k : ℚ) + 1) ≠ 0 := by
    exact denom_nonzero_from_recurrence m n h_recurrence
  have hsucc_nonzero : ∀ k, (n (k + 1) : ℚ) ≠ 0 := by
    exact n_succ_nonzero m n h_recurrence
  have hsucc_one := by exact fun k => succ_equal_implies_ratio_one m n hsucc_nonzero k
  have heq_prev := by exact fun k => ratio_one_implies_eq_prev m n (fun k => (hden k).1) k

  have hmn : ∀ k, m k ≠ n k := by
    intro k
    induction k with
    | zero => exact h_distinct
    | succ k ih =>
      by_contra h_eq_succ
      have hL : (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = 1 := by
        apply (hsucc_one k)
        exact h_eq_succ
      have hR : ((2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1) = 1) := by
        calc
          (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1) = (m (k + 1) : ℚ) / (n (k + 1) : ℚ) := Eq.symm (h_recurrence k)
          _ = 1 := hL
      have mk_eq_nk : m k = n k := by
        apply (heq_prev k)
        exact hR
      exact ih mk_eq_nk

  intro k
  intro hzero
  have eq_nat : m k = n k := by
    have : (m k : ℤ) - (n k : ℤ) = 0 := hzero
    exact (int_sub_eq_zero_iff_eq_nat (m k) (n k)).1 this
  exact (hmn k) eq_nat

theorem gcd_divides_both (p a b : ℕ) : p ∣ Nat.gcd a b → p ∣ a ∧ p ∣ b := by
  intro h
  have h₁ : p ∣ Nat.gcd a b := h
  have h₂ : p ∣ a := Nat.dvd_trans h₁ (Nat.gcd_dvd_left a b)
  have h₃ : p ∣ b := Nat.dvd_trans h₁ (Nat.gcd_dvd_right a b)
  exact ⟨h₂, h₃⟩

lemma odd_gcd {a b : ℕ} (ha : Odd a) (hb : Odd b) : Odd (Nat.gcd a b) := by

  by_contra hneg
  have h2 : 2 ∣ Nat.gcd a b := by
    exact not_odd_imp_two_dvd hneg
  have hdivs : 2 ∣ a ∧ 2 ∣ b := by
    exact (gcd_divides_both 2 a b) h2
  rcases hdivs with ⟨ha2, hb2⟩
  have na := odd_implies_not_two_dvd ha
  exact na ha2

lemma gcd_odd_of_odds {a b : ℕ} (ha : a % 2 = 1) (hb : b % 2 = 1) :
  (Nat.gcd a b) % 2 = 1 := by
  have ha_odd : Odd a := by
    exact (mod2_eq_one_iff_odd a).mp ha
  have hb_odd : Odd b := by
    exact (mod2_eq_one_iff_odd b).mp hb

  have h_gcd_odd : Odd (Nat.gcd a b) := by
    exact odd_gcd ha_odd hb_odd

  exact odd_iff_mod2_eq_one (Nat.gcd a b) h_gcd_odd

lemma odd_int_divides_cancel_two {d : ℕ} {z : ℤ}
  (hd_odd : d % 2 = 1) (hdiv : (d : ℤ) ∣ ((2 : ℤ) * z)) :
  (d : ℤ) ∣ z := by

  have hc : Nat.Coprime d 2 := by
    exact nat_coprime_of_odd d hd_odd
  have hbez_ex : ∃ x y : ℤ, x * (d : ℤ) + y * (2 : ℤ) = 1 := by
    exact int_bezout_of_coprime d 2 hc
  rcases hbez_ex with ⟨x, y, hbez⟩

  have hz_eq : z = x * (d : ℤ) * z + y * (2 : ℤ) * z := by
    calc
      z = 1 * z := by simp
      _ = (x * (d : ℤ) + y * (2 : ℤ)) * z := by rw [hbez]
      _ = x * (d : ℤ) * z + y * (2 : ℤ) * z := by ring

  have hterm1 : (d : ℤ) ∣ x * (d : ℤ) * z := by
    use x * z
    ring
  have hterm2 : (d : ℤ) ∣ y * (2 : ℤ) * z := by
    have h := int_mul_dvd_of_dvd (d : ℤ) hdiv (y : ℤ)
    simpa [mul_assoc] using h

  have hsum : (d : ℤ) ∣ (x * (d : ℤ) * z + y * (2 : ℤ) * z) := by
    exact int_dvd_add_of_dvd (d : ℤ) hterm1 hterm2

  rwa [← hz_eq] at hsum
lemma gcd_odd_and_dvd_diff
  (m n : ℕ → ℕ)
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  d % 2 = 1 ∧ ((d : ℤ) ∣ ((m k : ℤ) - (n k : ℤ))) := by
  let a := 2 * m k + 1
  let b := 2 * n k + 1
  let d := Nat.gcd a b

  have ha : a % 2 = 1 := by
    exact odd_two_mul_add_one_nat (m k)
  have hb : b % 2 = 1 := by
    exact odd_two_mul_add_one_nat (n k)

  have hd_odd : d % 2 = 1 := by
    exact gcd_odd_of_odds ha hb

  have hd_div_ab : (d : ℤ) ∣ ((a : ℤ) - (b : ℤ)) := by
    exact (gcd_cast_dvd_sub a b)

  have ab_eq_two_mul_diff : ((a : ℤ) - (b : ℤ)) = ((2 : ℤ) * ((m k : ℤ) - (n k : ℤ))) := by
    calc
      (a : ℤ) - (b : ℤ) = ((2 * m k + 1 : ℕ) : ℤ) - ((2 * n k + 1 : ℕ) : ℤ) := by rfl
      _ = ((2 : ℤ) * (m k : ℤ) + 1) - ((2 : ℤ) * (n k : ℤ) + 1) := by norm_cast
      _ = (2 : ℤ) * ((m k : ℤ) - (n k : ℤ)) := by ring

  have hd_div_two_diff : (d : ℤ) ∣ ((2 : ℤ) * ((m k : ℤ) - (n k : ℤ))) := by
    rw [ab_eq_two_mul_diff] at hd_div_ab
    exact hd_div_ab

  have hd_div_diff : (d : ℤ) ∣ ((m k : ℤ) - (n k : ℤ)) := by
    exact odd_int_divides_cancel_two hd_odd hd_div_two_diff

  exact ⟨hd_odd, hd_div_diff⟩

theorem D_succ_eq (t : ℕ → ℕ) (K : ℕ) :
  (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card =
    (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card +
      (if t (K + 1) < t K then 1 else 0) := by
  split_ifs with h
  case pos =>
    -- Case: t (K + 1) < t K is true
    have h₁ : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card =
      (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card + 1 := by
      -- We need to show that the number of inversions involving K+1 is 1 more than the number of inversions not involving K+1
      have h₂ : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card =
        (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card + 1 := by
        -- We need to show that the number of inversions involving K+1 is 1 more than the number of inversions not involving K+1
        have h₃ : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))) =
          (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)) ∪ {K} := by
          ext x
          simp [Finset.mem_filter, Finset.mem_range, Nat.lt_succ_iff]
          <;>
          by_cases h₄ : x < K <;>
          by_cases h₅ : t (x + 1) < t x <;>
          by_cases h₆ : x = K <;>
          simp_all [Nat.lt_succ_iff] <;>
          omega
        rw [h₃]
        simp [Finset.card_range]
        <;>
        omega
      exact h₂
    simp_all
  case neg =>
    -- Case: t (K + 1) < t K is false
    have h₁ : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card =
      (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card := by
      -- We need to show that the number of inversions involving K+1 is the same as the number of inversions not involving K+1
      have h₂ : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))) =
        (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)) := by
        ext x
        simp [Finset.mem_filter, Finset.mem_range, Nat.lt_succ_iff]
        <;>
        by_cases h₃ : x < K <;>
        by_cases h₄ : t (x + 1) < t x <;>
        by_cases h₅ : x = K <;>
        simp_all [Nat.lt_succ_iff] <;>
        omega
      rw [h₂]
    simp_all
lemma prefix_budget_induction (t : ℕ → ℕ) (hnoninc : ∀ k, t (k + 1) ≤ t k) (K : ℕ) :
  t K + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ t 0 := by
  induction K with
  | zero =>
    have : (Finset.filter (fun i => i < 0 ∧ t (i + 1) < t i) (Finset.range 0)).card = 0 := by
      simp [Finset.range_zero]
    simp [this]
  | succ K ih =>
    have Drec : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card
      = (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card +
        (if t (K + 1) < t K then 1 else 0) := by
      exact D_succ_eq t K

    rw [Drec]

    by_cases hdrop : t (K + 1) < t K
    ·
      rw [if_pos hdrop]
      let Dk := (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
      have : t (K + 1) + (Dk + 1) = (t (K + 1) + 1) + Dk := by
        rw [add_comm Dk 1, ← add_assoc]
      rw [this]
      have gap := gap_at_least_one t K hdrop
      have : (t (K + 1) + 1) + Dk ≤ t K + Dk := by
        apply Nat.add_le_add_right (Nat.le_trans gap (le_refl _)) Dk
      have ih' : t K + Dk ≤ t 0 := by
        exact ih
      exact Nat.le_trans this ih'
    ·
      rw [if_neg hdrop]
      have eq := no_drop_step t hnoninc K hdrop
      rw [eq]
      exact ih
lemma budget_inequality_on_prefix (t : ℕ → ℕ) (hnoninc : ∀ k, t (k + 1) ≤ t k) (K : ℕ) :
  (t K) + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ t 0 := by
  have hpref : t K + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
    ≤ t 0 := by
    exact prefix_budget_induction t hnoninc K
  exact hpref
lemma sum_budget_bound (t : ℕ → ℕ) (N : ℕ) (h0 : t 0 ≤ N)
  (hnoninc : ∀ k, t (k + 1) ≤ t k) :
  ∀ K, (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ t 0 := by
  intro K
  have hprefix : (t K) + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
    ≤ t 0 := by
    exact budget_inequality_on_prefix t hnoninc K

  calc
    (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
        ≤ (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card + (t K) :=
      Nat.le_add_right _ _
    _ = (t K) + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card := by
      rw [Nat.add_comm]
    _ ≤ t 0 := hprefix
lemma fin_card_bound (t : ℕ → ℕ) (N : ℕ) (h0 : t 0 ≤ N)
  (hnoninc : ∀ k, t (k + 1) ≤ t k) :
  ∀ K, (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ N := by

  (hF : ∀ x ∈ F, P x) :
  ∃ K, F ⊆ (Finset.filter (fun i => i < K ∧ P i) (Finset.range K)) := by
  use F.sup (fun i => i) + 1
  intro x hx
  have h1 : x ∈ F := hx
  have h2 : x < F.sup (fun i => i) + 1 := by
    have h3 : x ≤ F.sup (fun i => i) := by
      apply Finset.le_sup h1

  ((2 : ℕ) ^ u ∣ A) → u ≤ A := by

  have h₁ : 2 ^ u ∣ A := h

    have h₄ : 2 ^ u > A := by
      have h₅ : 2 ^ u ≥ 2 ^ A := by
        apply Nat.pow_le_pow_of_le_right
        linarith

      have h₆ : 2 ^ A > A := by
        apply Nat.lt_pow_self
        linarith
      linarith

    have h₆ : 2 ^ u ≤ A := Nat.le_of_dvd (by linarith) h₅
    linarith
  exact h₂

  (hpos : 0 < A)

  ¬ ((2 : ℕ) ^ (u + 1) ∣ A) := by
  intro hdiv_next
  have hbound : (u + 1) ≤ A := by
    exact pow_two_divides_bounded_by_A A (u + 1) hpos hdiv_next
  have in_range : u + 1 ∈ Finset.range (A + 1) := by
    exact Finset.mem_range.mpr (Nat.lt_succ_of_le hbound)
  have in_S : u + 1 ∈ S := by
    rw [hS_def]
    apply Finset.mem_filter.2
    constructor
    · exact in_range
    · exact hdiv_next
  have le_contra : u + 1 ≤ u := u_max (u + 1) in_S
  have : u < u := Nat.lt_of_succ_le le_contra
  exact Nat.lt_irrefl u this

    use u0
    simp [S, u0_in_range, hu0_div]

  have hu_div : ((2 : ℕ) ^ u ∣ A) := by
    simp [S] at u_mem
    exact u_mem.2

  have h₅ : 2 ^ (p + 1) ∣ 2 ^ p * x := h₄

      rw [Nat.pow_succ] at h₁₀
      simp [Nat.mul_dvd_mul_iff_left] at h₁₀

  (h : (2 : ℕ) ^ p * x = (2 : ℕ) ^ q * y) (hx_odd : x % 2 = 1) :

      exact pow_succ_dvd_pow_mul_implies_two_dvd hpow'

    exact False.elim hfalse
  ·

  p = q ∧ x = y := by

    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  {k : ℕ | ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)}.Finite := by
  classical

  have hΔ : ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
    exact diff_ne_zero m n h_distinct h_recurrence

  rcases exists_odd_measure m n hΔ h_recurrence h_coprime with ⟨t, ⟨htpos, hnoninc, hdec_step⟩⟩

  let bad : ℕ → Prop := fun k => ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)

  have hfinite : {k : ℕ | bad k}.Finite := by
    exact finite_of_strict_decrease_on_bad t htpos bad hnoninc hdec_step

  simpa [bad] using hfinite
