/-- Helper: combine the various equalities to identify the next odd part as the quotient r1 / d.
    We require an explicit witness r1' with r1 = d * r1'. -/
lemma odd_part_next_eq_quotient
  (A d u1 r1 u2 r2 r1' : ℕ)
  (hA : A = (2 : ℕ) ^ u1 * r1)
  (hr1_odd : r1 % 2 = 1) (hr1_pos : 0 < r1)
  (hr1_eq : r1 = d * r1')
  (hB_eq : 2 * (A / d) = (2 : ℕ) ^ u2 * r2)
  (hr2_odd : r2 % 2 = 1) (hr2_pos : 0 < r2)
  (hd_odd : d % 2 = 1) :
  r2 = r1' := by
  -- 1) from oddness of d get `1 ≤ d` so we can perform exact division on naturals
  have h1_le_d : 1 ≤ d := by
    exact odd_implies_one_le d hd_odd

  -- 2) rewrite A using `hA` and `hr1_eq` to expose a factor `d`
  have hA_mul : A = d * ((2 : ℕ) ^ u1 * r1') := by
    calc
      A = (2 : ℕ) ^ u1 * r1 := hA
      _ = (2 : ℕ) ^ u1 * (d * r1') := by rw [hr1_eq]
      _ = d * ((2 : ℕ) ^ u1 * r1') := by ring

  -- 3) perform exact division: A / d = (2^u1) * r1'
  have hdiv : A / d = (2 : ℕ) ^ u1 * r1' := by
    exact nat_div_eq_of_mul_eq h1_le_d hA_mul

  -- 4) rewrite 2 * (A / d) as (2^(u1+1)) * r1'
  have hleft : 2 * (A / d) = (2 : ℕ) ^ (u1 + 1) * r1' := by
    calc
      2 * (A / d) = 2 * ((2 : ℕ) ^ u1 * r1') := by rw [hdiv]
      _ = (2 : ℕ) ^ (u1 + 1) * r1' := by ring

  -- 5) deduce r1' is odd from r1 = d * r1' and oddness of r1 and d
  have hr1'_odd : r1' % 2 = 1 := by
    exact odd_quotient_of_odd_divisor r1 d r1' hr1_eq hr1_odd hd_odd

  -- 6) combine the two expressions for 2 * (A / d) to get a two-adic equality
  have h_eq_pow : (2 : ℕ) ^ (u1 + 1) * r1' = (2 : ℕ) ^ u2 * r2 := by
    calc
      (2 : ℕ) ^ (u1 + 1) * r1' = 2 * (A / d) := by rw [← hleft]
      _ = (2 : ℕ) ^ u2 * r2 := by exact hB_eq

  -- 7) apply uniqueness of two-adic decomposition for odd parts to conclude r1' = r2
  have huniq := pow_mul_odd_eq_pow_mul_odd (u1 + 1) u2 r1' r2 h_eq_pow hr1'_odd hr2_odd
  rcases huniq with ⟨_heq_pow, heq_oddpart⟩
  -- `heq_oddpart` gives r1' = r2, return the symmetric form required by the statement
  exact Eq.symm heq_oddpart
