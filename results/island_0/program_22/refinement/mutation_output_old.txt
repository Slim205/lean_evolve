lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  (m (k + 1) = (2 * m k + 1) / d) ∧ (n (k + 1) = (2 * n k + 1) / d) := by
  -- define the gcd used in the conclusion
  let d := Nat.gcd (2 * m k + 1) (2 * n k + 1)

  -- obtain the successor coprimality from the hypothesis
  have hsucc_copr : Nat.Coprime (m (k + 1)) (n (k + 1)) := by
    exact h_coprime (k + 1) (Nat.succ_pos k)

  -- apply the reduced-fraction uniqueness helper to the given rational equality
  have hred : let g := Nat.gcd (2 * m k + 1) (2 * n k + 1) in
    m (k + 1) = (2 * m k + 1) / g ∧ n (k + 1) = (2 * n k + 1) / g := by
    exact reduced_fraction_gcd_quotients (hsucc_copr) (h_recurrence k)

  -- the `g` in the helper is definitionally the same as our `d`, so we can close the goal
  exact hred
