/-- Helper (abstract): if A = 2^u1 * r1 with r1 odd > 0, and `d` is an odd divisor of `A`,
    and if we form B := 2 * (A / d), and B admits a decomposition B = 2^u2 * r2 with r2 odd > 0,
    then r2 ≤ r1. The extra hypothesis `d ∣ A` is essential. -/
lemma odd_part_le_from_div_by_odd (A d : ℕ) (hdvd : d ∣ A) (u1 r1 u2 r2 : ℕ)
  (hA : A = (2 : ℕ) ^ u1 * r1) (hr1_odd : r1 % 2 = 1) (hr1_pos : 0 < r1)
  (hB_eq : 2 * (A / d) = (2 : ℕ) ^ u2 * r2) (hr2_odd : r2 % 2 = 1) (hr2_pos : 0 < r2)
  (hd_odd : d % 2 = 1) :
  r2 ≤ r1 := by
  -- 1) d is coprime to 2^u1
  have hcop : Nat.Coprime d ((2 : ℕ) ^ u1) := by
    exact odd_implies_coprime_pow_two d u1 hd_odd

  -- 2) from A = 2^u1 * r1 and d ∣ A we get d ∣ (2^u1 * r1); apply coprime cancellation to deduce d ∣ r1
  have hd_dvd_r1 : d ∣ r1 := by
    have : d ∣ ( (2 : ℕ) ^ u1 * r1 ) := by
      rw [hA]
      exact hdvd
    exact coprime_dvd_of_dvd_mul_right hcop this

  -- 3) witness the quotient r1' with r1 = d * r1'
  rcases nat_dvd_exists_mul d r1 hd_dvd_r1 with ⟨r1', hr1_eq⟩

  -- 4) rewrite A = d * (2^u1 * r1') and use exact division to get A / d = 2^u1 * r1'
  have hA_mul : A = d * ((2 : ℕ) ^ u1 * r1') := by
    calc
      A = (2 : ℕ) ^ u1 * r1 := hA
      _ = (2 : ℕ) ^ u1 * (d * r1') := by rw [hr1_eq]
      _ = d * ((2 : ℕ) ^ u1 * r1') := by ring
  have h_divA : A / d = (2 : ℕ) ^ u1 * r1' := by
    exact nat_div_eq_of_mul_eq hA_mul

  -- 5) rewrite 2 * (A / d) as 2^(u1+1) * r1'
  have h_left : (2 : ℕ) ^ (u1 + 1) * r1' = 2 * (A / d) := by
    calc
      (2 : ℕ) ^ (u1 + 1) * r1' = 2 * ((2 : ℕ) ^ u1 * r1') := by ring
      _ = 2 * (A / d) := by rw [h_divA]

  -- 6) combine with the given decomposition of B to get equality of two 2-adic decompositions
  have h_comb : (2 : ℕ) ^ (u1 + 1) * r1' = (2 : ℕ) ^ u2 * r2 := by
    calc
      (2 : ℕ) ^ (u1 + 1) * r1' = 2 * (A / d) := by exact h_left
      _ = (2 : ℕ) ^ u2 * r2 := by exact hB_eq

  -- 7) show r1' is odd from r1 = d * r1' and oddness of r1 and d
