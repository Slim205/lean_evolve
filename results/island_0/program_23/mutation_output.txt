lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  (m (k + 1) = (2 * m k + 1) / d) ∧ (n (k + 1) = (2 * n k + 1) / d) := by
  -- Cross-multiply the rational recurrence to obtain the integral relation
  have heq_nat : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
    exact cross_mul_from_recurrence m n h_recurrence k

  -- use coprimality of the successor pair (k+1 > 0)
  have hcop_succ : Nat.Coprime (m (k + 1)) (n (k + 1)) := by
    exact h_coprime (k + 1) (Nat.succ_pos k)

  -- from coprimality and the cross-multiplication deduce divisibilities
  have hdivs : (m (k + 1) ∣ (2 * m k + 1)) ∧ (n (k + 1) ∣ (2 * n k + 1)) := by
    have := coprime_mul_eq_imp_dvd hcop_succ heq_nat
    exact this

  rcases hdivs with ⟨hm_div, hn_div⟩

  -- apply the reduced-fraction helper to conclude the gcd-quotient representation
  have hquot := reduced_fraction_gives_quotients hcop_succ hm_div hn_div heq_nat
  rcases hquot with ⟨h_m_eq, h_n_eq⟩

  -- set local d := gcd (2*m_k+1) (2*n_k+1) and return the conjunction
  let d := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  -- the helper returns equalities in terms of the same `g = gcd c d` we just defined,
  -- so we can return them directly
  exact ⟨h_m_eq, h_n_eq⟩
