  (hA : A = (2 : ℕ) ^ u * r) (hr_odd : r % 2 = 1) (hd_odd : d % 2 = 1) (hdvd : d ∣ A) :
  d ∣ r := by
  -- 1) d is coprime to 2^u because d is odd
  have hcop : Nat.Coprime d ((2 : ℕ) ^ u) := by
    exact odd_implies_coprime_pow_two d u hd_odd
