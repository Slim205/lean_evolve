lemma natAbs_two_mul_div_of_odd_divisor (z : ℤ) (d : ℕ)
  (hd_odd : d % 2 = 1) (hdiv : (d : ℤ) ∣ z) :
  Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = 2 * (Int.natAbs z / d) := by
  -- 1) obtain nonzeroness of (d : ℤ)
  have hd_ne : (d : ℤ) ≠ 0 := by
    exact odd_nat_nonzero d hd_odd

  -- 2) get explicit quotient q with z = (d : ℤ) * q
  have hex : ∃ q : ℤ, z = (d : ℤ) * q := by
    exact int_dvd_exists_quot d z hdiv
  rcases hex with ⟨q, hq⟩

  -- 3) rewrite the left-hand side using exact division/cancellation
  have h_div_cancel : ((2 : ℤ) * z) / (d : ℤ) = (2 : ℤ) * q := by
    calc
      ((2 : ℤ) * z) / (d : ℤ) = ((2 : ℤ) * ((d : ℤ) * q)) / (d : ℤ) := by rw [hq]
      _ = (2 : ℤ) * q := by exact int_div_mul_cancel (d : ℤ) q hd_ne

  -- 4) apply Int.natAbs to both sides and use multiplicativity for 2
  have h_lhs_natabs : Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = Int.natAbs ((2 : ℤ) * q) := by
    congr
    exact h_div_cancel

  have h_natabs_two : Int.natAbs ((2 : ℤ) * q) = 2 * Int.natAbs q := by
    exact natAbs_mul_two q

  -- 5) relate Int.natAbs z / d to Int.natAbs q
  have h_natabs_div : Int.natAbs z / d = Int.natAbs q := by
    -- from z = d * q we get Int.natAbs z = d * Int.natAbs q, so division yields equality
    have : Int.natAbs ((d : ℤ) * q) / d = Int.natAbs q := by
      exact natAbs_div_exact d q rfl
    -- rewrite using hq
    calc
      Int.natAbs z / d = Int.natAbs ((d : ℤ) * q) / d := by rw [hq]
      _ = Int.natAbs q := this

  -- 6) assemble the final equality
  calc
    Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = Int.natAbs ((2 : ℤ) * q) := by rw [h_lhs_natabs]
    _ = 2 * Int.natAbs q := by rw [h_natabs_two]
    _ = 2 * (Int.natAbs z / d) := by rw [← h_natabs_div]
