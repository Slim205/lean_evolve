lemma pow_mul_odd_eq_pow_mul_odd (p q x y : ℕ)
  (h : (2 : ℕ) ^ p * x = (2 : ℕ) ^ q * y) (hx_odd : x % 2 = 1) (hy_odd : y % 2 = 1) :
  p = q ∧ x = y := by
  -- Use the auxiliary lemmas to compare exponents and cancel.
  have hle1 : p ≤ q := by
    exact le_of_mul_eq_left p q x y h hx_odd
  -- swap sides to get the reverse inequality
  have hle2 : q ≤ p := by
    -- apply the same lemma to the symmetric equality
    exact le_of_mul_eq_left q p y x (Eq.symm h) hy_odd
  have heq_pq : p = q := by
    exact Nat.le_antisymm hle1 hle2
  -- cancel the common power to obtain x = y
  have hxy : x = y := by
    have hpow : (2 : ℕ) ^ p * x = (2 : ℕ) ^ p * y := by
      rw [heq_pq] at h
      exact h
    exact cancel_pow_mul_eq p x y hpow
  exact ⟨heq_pq, hxy⟩
