lemma cross_mul_eq_nat
  (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) =
      ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (k : ℕ)
  (h_n_succ_nonzero_k : (n (k + 1) : ℚ) ≠ 0)
  (h_two_nk_nonzero : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0) :
  m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
  -- obtain the product equality in ℚ from the recurrence (clearing denominators in ℚ)
  have hq : (m (k + 1) : ℚ) * ((2 : ℚ) * (n k : ℚ) + 1) =
            (n (k + 1) : ℚ) * ((2 : ℚ) * (m k : ℚ) + 1) := by
    exact recurrence_cross_Q m n k h_recurrence h_n_succ_nonzero_k h_two_nk_nonzero

  -- rewrite the numeric factors as casts of naturals so we have products of casts
  have hleft_cast : ((2 : ℚ) * (n k : ℚ) + 1) = ((2 * n k + 1 : ℕ) : ℚ) := by
    exact cast_two_mul_nat_plus_one (n k)
  have hright_cast : ((2 : ℚ) * (m k : ℚ) + 1) = ((2 * m k + 1 : ℕ) : ℚ) := by
    exact cast_two_mul_nat_plus_one (m k)

  -- rewrite the ℚ-equality to an equality of products of natural casts
  have hq_natcasts : (m (k + 1) : ℚ) * ((2 * n k + 1 : ℕ) : ℚ) =
                     (n (k + 1) : ℚ) * ((2 * m k + 1 : ℕ) : ℚ) := by
    rw [hleft_cast, hright_cast] at hq
    exact hq

  -- apply injection from rational casts to natural equality of products
  have hnat : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
    apply nat_cast_inj_mul
    exact hq_natcasts

  exact hnat

/-- Main lemma: extract a common natural scaling factor `c` from the rational recurrence. -/
lemma next_scaling_exists
  (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) =
      ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (h_n_succ_nonzero : ∀ k, (n (k + 1) : ℚ) ≠ 0)
  (h_coprime_succ : ∀ k, Nat.Coprime (m (k + 1)) (n (k + 1)))
  (k : ℕ) :
  ∃ c : ℕ, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
  -- nonzeroness of (2 * n_k + 1) as rationals
  have h2n_nonzero : ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 := by
    exact two_mul_nat_plus_one_nonzero (n k)

  -- cross-multiply the rational equality to get the integral equality
  have hcross : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
    exact cross_mul_eq_nat m n h_recurrence k (h_n_succ_nonzero k) h2n_nonzero

  -- from the cross equality we see that `m_{k+1}` divides `n_{k+1}*(2*m_k+1)`; but the
  -- left-hand side also equals `m_{k+1}*(2*n_k+1)`, hence `m_{k+1} ∣ n_{k+1}*(2*m_k+1)`.
  have hm_div_mul : m (k + 1) ∣ n (k + 1) * (2 * m k + 1) := by
    have : n (k + 1) * (2 * m k + 1) = m (k + 1) * (2 * n k + 1) := by
      exact Eq.symm hcross
    rw [this]
    exact dvd_mul_right _ _

  -- use coprimality to deduce `m_{k+1} ∣ (2*m_k+1)`
  have hm_div : m (k + 1) ∣ (2 * m k + 1) := by
    exact @coprime_dvd_of_dvd_mul_right (m (k + 1)) (n (k + 1)) (2 * m k + 1) (h_coprime_succ k) hm_div_mul

  -- similarly for the n-side: `n_{k+1} ∣ (2*n_k+1)`
  have hn_div_mul : n (k + 1) ∣ m (k + 1) * (2 * n k + 1) := by
    have : m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
      exact hcross
    rw [this]
    exact dvd_mul_right _ _
  have hn_div : n (k + 1) ∣ (2 * n k + 1) := by
    exact @coprime_dvd_of_dvd_mul_right (n (k + 1)) (m (k + 1)) (2 * n k + 1) (Nat.Coprime.symm (h_coprime_succ k)) hn_div_mul

  -- finally build the common scaling factor `c` from these divisibilities and the cross equality
  have hex : ∃ c, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
    exact build_common_scaling_from_divs m n k hcross hm_div hn_div

  exact hex

lemma next_eq_div_gcd
  (m n : ℕ → ℕ)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) =
      ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  ((2 * m k + 1) = d * m (k + 1)) ∧ ((2 * n k + 1) = d * n (k + 1)) := by
  -- obtain a scaling factor c making the two numerators multiples of the successors
  have hscal : ∃ c, (2 * m k + 1) = c * m (k + 1) ∧ (2 * n k + 1) = c * n (k + 1) := by
    -- supply the auxiliary nonzeroness and coprimality hypotheses required by
    -- `next_scaling_exists`.
    exact next_scaling_exists m n h_recurrence (n_succ_nonzero m n h_recurrence)
      (fun i => h_coprime (i + 1) (Nat.succ_pos i)) k
  rcases hscal with ⟨c, ⟨hc1, hc2⟩⟩

  -- use coprimality of the successor pair (applied at k+1) to identify c with the gcd
  have hc_gcd : c = Nat.gcd (2 * m k + 1) (2 * n k + 1) := by
    have succ_copr : Nat.Coprime (m (k + 1)) (n (k + 1)) := h_coprime (k + 1) (Nat.succ_pos k)
    exact scaling_factor_eq_gcd m n k c hc1 hc2 succ_copr

  -- substitute c by the gcd and return the multiplicative equalities (exact divisibility form)
  let d := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  have : (2 * m k + 1) = d * m (k + 1) := by
    rw [hc_gcd] at hc1
    exact hc1
  have : (2 * n k + 1) = d * n (k + 1) := by
    rw [hc_gcd] at hc2
    exact hc2
  -- assemble the conjunction
  exact ⟨by assumption, by assumption⟩

/-- Step 2: the differences never become zero. -/
lemma diff_ne_zero
  (m n : ℕ → ℕ)
  (h_distinct : m 0 ≠ n 0)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
  -- instantiate helpers (all are `sorry` above)
  have hden : ∀ k, ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 ∧ ((2 : ℚ) * (m k : ℚ) + 1) ≠ 0 := by
    exact denom_nonzero_from_recurrence m n h_recurrence
  have hsucc_nonzero : ∀ k, (n (k + 1) : ℚ) ≠ 0 := by
    exact n_succ_nonzero m n h_recurrence
  have hsucc_one := by exact fun k => succ_equal_implies_ratio_one m n hsucc_nonzero k
  have heq_prev := by exact fun k => ratio_one_implies_eq_prev m n (fun k => (hden k).1) k

  -- prove by induction that m k ≠ n k for all k
  have hmn : ∀ k, m k ≠ n k := by
    intro k
    induction k with
    | zero => exact h_distinct
    | succ k ih =>
      by_contra h_eq_succ
      -- if m_{k+1} = n_{k+1} then the left ratio is 1
      have hL : (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = 1 := by
        apply (hsucc_one k)
        exact h_eq_succ
      -- by recurrence the right-hand ratio equals the left, so the right is 1
      have hR : ((2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1) = 1) := by
        calc
          (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1) = (m (k + 1) : ℚ) / (n (k + 1) : ℚ) := Eq.symm (h_recurrence k)
          _ = 1 := hL
      -- hence m_k = n_k, contradicting IH
      have mk_eq_nk : m k = n k := by
        apply (heq_prev k)
        exact hR
      exact ih mk_eq_nk

  -- finish: translate nat-inequality to integer-subtraction nonzeroness using helper
  intro k
  intro hzero
  have eq_nat : m k = n k := by
    have : (m k : ℤ) - (n k : ℤ) = 0 := hzero
    exact (int_sub_eq_zero_iff_eq_nat (m k) (n k)).1 this
  exact (hmn k) eq_nat

/-- If `a` and `b` are odd then `gcd a b` is odd. -/
lemma odd_gcd {a b : ℕ} (ha : Odd a) (hb : Odd b) : Odd (Nat.gcd a b) := by
  -- Proof by contradiction: if the gcd were not odd then 2 divides it, hence divides both a and b,
  -- contradicting their oddness. We glue the argument using the helper lemmas above.
  by_contra hneg
  -- from ¬Odd (gcd a b) get 2 ∣ gcd a b
  have h2 : 2 ∣ Nat.gcd a b := by
    exact not_odd_imp_two_dvd hneg
  -- hence 2 divides both a and b
  have hdivs : 2 ∣ a ∧ 2 ∣ b := by
    exact (gcd_divides_both 2 a b) h2
  rcases hdivs with ⟨ha2, hb2⟩
  -- but a is odd, so 2 ∤ a, contradiction
  have na := odd_implies_not_two_dvd ha
  exact na ha2

lemma odd_int_divides_cancel_two {d : ℕ} {z : ℤ}
  (hd_odd : d % 2 = 1) (hdiv : (d : ℤ) ∣ ((2 : ℤ) * z)) :
  (d : ℤ) ∣ z := by
  -- Strategy (glue): 1) get `Nat.Coprime d 2` from oddness; 2) get Bézout x,y with x*d + y*2 = 1;
  -- 3) multiply that identity by z and use closure of divisibility under multiplication and addition.
  have hc : Nat.Coprime d 2 := by
    exact nat_coprime_of_odd d hd_odd
  have hbez_ex : ∃ x y : ℤ, x * (d : ℤ) + y * (2 : ℤ) = 1 := by
    exact int_bezout_of_coprime d 2 hc
  rcases hbez_ex with ⟨x, y, hbez⟩

  -- multiply the Bézout identity by z : z = x*d*z + y*2*z
  have hz_eq : z = x * (d : ℤ) * z + y * (2 : ℤ) * z := by
    calc
      z = 1 * z := by simp
      _ = (x * (d : ℤ) + y * (2 : ℤ)) * z := by rw [hbez]
      _ = x * (d : ℤ) * z + y * (2 : ℤ) * z := by ring

  -- show each summand on RHS is divisible by (d : ℤ)
  have hterm1 : (d : ℤ) ∣ x * (d : ℤ) * z := by
    use x * z
    ring

  have hterm2 : (d : ℤ) ∣ y * (2 : ℤ) * z := by
    have h := int_mul_dvd_of_dvd (d : ℤ) hdiv (y : ℤ)
    simpa [mul_assoc] using h

  -- combine the two divisibilities to get d ∣ RHS, hence d ∣ z
  have hsum : (d : ℤ) ∣ (x * (d : ℤ) * z + y * (2 : ℤ) * z) := by
    exact int_dvd_add_of_dvd (d : ℤ) hterm1 hterm2

  -- finish by rewriting z as that RHS
  rwa [← hz_eq] at hsum

lemma gcd_odd_and_dvd_diff
  (m n : ℕ → ℕ)
  (k : ℕ) :
  let d : ℕ := Nat.gcd (2 * m k + 1) (2 * n k + 1)
  d % 2 = 1 ∧ ((d : ℤ) ∣ ((m k : ℤ) - (n k : ℤ))) := by
  -- abbreviate the two relevant odd naturals
  let a := 2 * m k + 1
  let b := 2 * n k + 1
  let d := Nat.gcd a b

  -- each of a and b is odd
  have ha : a % 2 = 1 := by
    -- reduce to the simple lemma for 2*x+1
    exact odd_two_mul_add_one_nat (m k)
  have hb : b % 2 = 1 := by
    exact odd_two_mul_add_one_nat (n k)

  -- gcd of two odd numbers is odd
  have hd_odd : d % 2 = 1 := by
    exact gcd_odd_of_odds ha hb

  -- gcd (cast to ℤ) divides a - b in ℤ
  have hd_div_ab : (d : ℤ) ∣ ((a : ℤ) - (b : ℤ)) := by
    exact (gcd_cast_dvd_sub a b)

  -- rewrite a - b as 2 * (m k - n k) in ℤ
  have ab_eq_two_mul_diff : ((a : ℤ) - (b : ℤ)) = ((2 : ℤ) * ((m k : ℤ) - (n k : ℤ))) := by
    -- expand definitions and reduce casts, then finish by ring
    calc
      (a : ℤ) - (b : ℤ) = ((2 * m k + 1 : ℕ) : ℤ) - ((2 * n k + 1 : ℕ) : ℤ) := by rfl
      _ = ((2 : ℤ) * (m k : ℤ) + 1) - ((2 : ℤ) * (n k : ℤ) + 1) := by norm_cast
      _ = (2 : ℤ) * ((m k : ℤ) - (n k : ℤ)) := by ring

  -- hence gcd divides 2 * (m k - n k)
  have hd_div_two_diff : (d : ℤ) ∣ ((2 : ℤ) * ((m k : ℤ) - (n k : ℤ))) := by
    rw [ab_eq_two_mul_diff] at hd_div_ab
    exact hd_div_ab

  -- cancel the factor 2 using oddness of d to deduce gcd divides (m k - n k)
  have hd_div_diff : (d : ℤ) ∣ ((m k : ℤ) - (n k : ℤ)) := by
    exact odd_int_divides_cancel_two hd_odd hd_div_two_diff

  -- assemble the conjunction: gcd is odd and divides the integer difference
  exact ⟨hd_odd, hd_div_diff⟩

/-- Helper: recursion for the drop-count on prefixes.
    D(K) := card (filter (fun i => i < K ∧ t(i+1) < t i) (range K)).
    Then D(K+1) = D(K) + (if t (K+1) < t K then 1 else 0). -/
lemma D_succ_eq (t : ℕ → ℕ) (K : ℕ) :
  (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card =
    (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card +
      (if t (K + 1) < t K then 1 else 0) := by
  -- combinatorial decomposition of `range (K+1)` as `range K` ∪ {K}; proof omitted
  sorry

/-- Prefix budget induction:
    for D(K) := #{i < K | t(i+1) < t i}, one has t K + D(K) ≤ t 0.
    The proof proceeds by induction on K, using `gap_at_least_one` on drop steps and
    `no_drop_step` on non-drop steps. The heavy combinatorial step is `D_succ_eq`. -/
lemma prefix_budget_induction (t : ℕ → ℕ) (hnoninc : ∀ k, t (k + 1) ≤ t k) (K : ℕ) :
  t K + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ t 0 := by
  induction K with
  | zero =>
    -- D(0) = 0 because `range 0` is empty
    have : (Finset.filter (fun i => i < 0 ∧ t (i + 1) < t i) (Finset.range 0)).card = 0 := by
      simp [Finset.range_zero]
    simp [this]
  | succ K ih =>
    -- rewrite the `K+1` case using the recursion for D
    have Drec : (Finset.filter (fun i => i < (K + 1) ∧ t (i + 1) < t i) (Finset.range (K + 1))).card
      = (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card +
        (if t (K + 1) < t K then 1 else 0) := by
      exact D_succ_eq t K

    -- reduce the goal using the D recursion
    rw [Drec]

    -- case split on whether there is a strict drop at index K
    by_cases hdrop : t (K + 1) < t K
    · -- drop case: D(K+1) = D(K) + 1 and one-step gap gives t (K+1) + 1 ≤ t K
      rw [if_pos hdrop]
      -- let Dk denote the card for the prefix K
      let Dk := (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
      -- rearrange the left-hand side: t (K+1) + (Dk + 1) = (t (K+1) + 1) + Dk
      have : t (K + 1) + (Dk + 1) = (t (K + 1) + 1) + Dk := by
        rw [add_comm Dk 1, ← add_assoc]
      rw [this]
      -- use the discrete gap lemma to consume one unit from t K
      have gap := gap_at_least_one t K hdrop
      have : (t (K + 1) + 1) + Dk ≤ t K + Dk := by
        apply Nat.add_le_add_right (Nat.le_trans gap (le_refl _)) Dk -- (t (K+1)+1) ≤ t K
      -- apply the inductive hypothesis on K: t K + Dk ≤ t 0
      have ih' : t K + Dk ≤ t 0 := by
        exact ih
      exact Nat.le_trans this ih'
    · -- no-drop case: D(K+1) = D(K) and t (K+1) = t K
      rw [if_neg hdrop]
      -- from nonincreasingness and ¬(t (K+1) < t K) we get equality
      have eq := no_drop_step t hnoninc K hdrop
      rw [eq]
      -- now the goal is exactly the IH
      exact ih

/-- Budget inequality on an initial prefix [0,K): the number of strict drops in `[0,K)` plus
`t K` is bounded by the initial value `t 0`. We glue the proof from the decomposition lemmas. -/
lemma budget_inequality_on_prefix (t : ℕ → ℕ) (hnoninc : ∀ k, t (k + 1) ≤ t k) (K : ℕ) :
  (t K) + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ t 0 := by
  -- delegate the core induction to the helper lemma
  have hpref : t K + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
    ≤ t 0 := by
    exact prefix_budget_induction t hnoninc K
  exact hpref

/-- Summed budget bound on initial segments.
    Intuitively, each strict drop in [0,K) consumes at least one unit from the initial
    value `t 0`, so the number of drops up to K is bounded by `t 0`. We implement the
    finishing glue using the helper `budget_inequality_on_prefix`. -/
lemma sum_budget_bound (t : ℕ → ℕ) (N : ℕ) (h0 : t 0 ≤ N)
  (hnoninc : ∀ k, t (k + 1) ≤ t k) :
  ∀ K, (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ t 0 := by
  intro K
  -- use the prefix budget inequality: t K + D(K) ≤ t 0
  have hprefix : (t K) + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
    ≤ t 0 := by
    exact budget_inequality_on_prefix t hnoninc K

  -- now D(K) ≤ t K + D(K) and by transitivity with the prefix bound we get D(K) ≤ t 0
  calc
    (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card
        ≤ (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card + (t K) :=
      Nat.le_add_right _ _
    _ = (t K) + (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card := by
      rw [Nat.add_comm]
    _ ≤ t 0 := hprefix

/-- Cardinality bound for the number of strict drops in initial segments.
    This is the wrapper glue-lemma that uses the more detailed `sum_budget_bound`
    above to conclude the desired uniform cardinality bound by transitivity with
    the global bound `t 0 ≤ N`. -/
lemma fin_card_bound (t : ℕ → ℕ) (N : ℕ) (h0 : t 0 ≤ N)
  (hnoninc : ∀ k, t (k + 1) ≤ t k) :
  ∀ K, (Finset.filter (fun i => i < K ∧ t (i + 1) < t i) (Finset.range K)).card ≤ N := by

    exact sum_budget_bound t N h0 hnoninc
  -- now finish by transitivity with the hypothesis t 0 ≤ N
  intro K
  exact Nat.le_trans (hsum K) h0

/-- Convert the uniform bound on prefix-cardinality into finiteness of the full set.
    The proof is by contradiction using the three helpers above. -/
lemma finite_of_card_bounded {P : ℕ → Prop} (N : ℕ) [DecidablePred P]
  (hcard : ∀ K, (Finset.filter (fun i => i < K ∧ P i) (Finset.range K)).card ≤ N) :
  {k : ℕ | P k}.Finite := by
  by_contra hnot_fin
  -- `hnot_fin` is exactly `Set.Infinite {k | P k}`
  have hinf : Set.Infinite {k | P k} := hnot_fin
  -- pick a finite subset of size N+1 from the infinite set
  rcases infinite_gives_large_finite_subset (N + 1) hinf with ⟨F, hFprop, hFcard⟩
  -- apply the prefix-card bound to get F.card ≤ N, contradicting F.card = N+1
  have hFle : F.card ≤ N := by
    exact card_le_from_prefix_bound N hcard F hFprop
  -- rewrite the equality into the inequality and close by contradiction
  have hbad : N + 1 ≤ N := by
    rw [hFcard] at hFle
    exact hFle
  exact Nat.not_succ_le_self N hbad

/-- Helper (4): if `S` is `filter (λ v, 2^v ∣ A) (range (A+1))`, `A > 0`, and `u` is a maximal
    element of `S` then `2^(u+1)` does not divide `A`. The extra `0 < A` hypothesis rules
    out the degenerate `A = 0` case (where every power of 2 divides `A`). -/
lemma maximality_implies_no_next_divides (A : ℕ) (S : Finset ℕ) (u : ℕ)
  (hpos : 0 < A)
  (hS_def : S = Finset.filter (fun v => ((2 : ℕ) ^ v ∣ A)) (Finset.range (A + 1)))
  (u_mem : u ∈ S) (u_max : ∀ v ∈ S, v ≤ u) :
  ¬ ((2 : ℕ) ^ (u + 1) ∣ A) := by
  -- assume contrary and derive that `u+1 ∈ S`, contradicting maximality
  intro hdiv_next
  -- use the bound lemma to see that (u+1) ≤ A, so u+1 ∈ range (A+1)
  have hbound : (u + 1) ≤ A := by
    exact pow_two_divides_bounded_by_A A (u + 1) hpos hdiv_next
  have in_range : u + 1 ∈ Finset.range (A + 1) := by
    exact Finset.mem_range.mpr (Nat.lt_succ_of_le hbound)
  -- hence u+1 ∈ S by definition of S
  have in_S : u + 1 ∈ S := by
    rw [hS_def]
    apply Finset.mem_filter.2
    constructor
    · exact in_range
    · exact hdiv_next
  -- maximality gives (u+1) ≤ u, contradiction
  have le_contra : u + 1 ≤ u := u_max (u + 1) in_S
  have : u < u := Nat.lt_of_succ_le le_contra
  exact Nat.lt_irrefl u this

/-- Construct the maximal exponent `u` with `2^u ∣ A` and `¬(2^(u+1) ∣ A)` by restricting to the
    finite set `Finset.filter (λ v, 2^v ∣ A) (range (A+1))` and taking its maximum. -/
lemma exists_max_pow_two_dividing (A : ℕ) (hpos : 0 < A) :
  ∃ u, ((2 : ℕ) ^ u ∣ A) ∧ ¬ ((2 : ℕ) ^ (u + 1) ∣ A) := by
  -- get some exponent u0 with 2^u0 ∣ A
  have hex := exists_two_pow_divides A hpos
  rcases hex with ⟨u0, hu0_div⟩

  -- bound u0 by A so that u0 ∈ range (A+1)
  have hub : u0 ≤ A := by
    exact pow_two_divides_bounded_by_A A u0 hpos hu0_div

  have u0_in_range : u0 ∈ Finset.range (A + 1) := by
    exact Finset.mem_range.mpr (Nat.lt_succ_of_le hub)

  -- consider the finite set of exponents ≤ A that divide A
  let S : Finset ℕ := Finset.filter (fun v => ((2 : ℕ) ^ v ∣ A)) (Finset.range (A + 1))

  -- S is nonempty because u0 ∈ S
  have S_nonempty : S.Nonempty := by
    use u0
    simp [S, u0_in_range, hu0_div]

  -- take a maximum element u of S
  rcases exists_max_in_finset S S_nonempty with ⟨u, ⟨u_mem, u_max⟩⟩

  -- u ∈ S gives the divisibility
  have hu_div : ((2 : ℕ) ^ u ∣ A) := by
    simp [S] at u_mem
    exact u_mem.2

  -- maximality in S yields ¬ 2^(u+1) ∣ A
  have hnot_next : ¬ ((2 : ℕ) ^ (u + 1) ∣ A) := by
    exact maximality_implies_no_next_divides A S u hpos rfl u_mem u_max

  exact ⟨u, hu_div, hnot_next⟩

/-- Helper (abstract): if A = 2^u1 * r1 with r1 odd > 0, and `d` is an odd divisor of `A`,
    and if we form B := 2 * (A / d), and B admits a decomposition B = 2^u2 * r2 with r2 odd > 0,
    then r2 ≤ r1. The extra hypothesis `d ∣ A` is essential. -/
lemma odd_part_le_from_div_by_odd (A d : ℕ) (hdvd : d ∣ A) (u1 r1 u2 r2 : ℕ)
  (hA : A = (2 : ℕ) ^ u1 * r1) (hr1_odd : r1 % 2 = 1) (hr1_pos : 0 < r1)
  (hB_eq : 2 * (A / d) = (2 : ℕ) ^ u2 * r2) (hr2_odd : r2 % 2 = 1) (hr2_pos : 0 < r2)
  (hd_odd : d % 2 = 1) :
  r2 ≤ r1 := by
  -- 1) d is coprime to 2^u1
  have hcop : Nat.Coprime d ((2 : ℕ) ^ u1) := by
    exact odd_implies_coprime_pow_two d u1 hd_odd

  -- 2) from A = 2^u1 * r1 and d ∣ A we get d ∣ (2^u1 * r1); apply coprime cancellation to deduce d ∣ r1
  have hd_dvd_r1 : d ∣ r1 := by
    have : d ∣ ( (2 : ℕ) ^ u1 * r1 ) := by
      rw [← hA]
      exact hdvd
    exact coprime_dvd_of_dvd_mul_right hcop this

  -- 3) witness the quotient r1' with r1 = d * r1'
  rcases nat_dvd_exists_mul d r1 hd_dvd_r1 with ⟨r1', hr1_eq⟩

  -- 4) rewrite A = d * (2^u1 * r1') and use exact division to get A / d = 2^u1 * r1'
  have hA_mul : A = d * ((2 : ℕ) ^ u1 * r1') := by
    calc
      A = (2 : ℕ) ^ u1 * r1 := hA
      _ = (2 : ℕ) ^ u1 * (d * r1') := by rw [hr1_eq]
      _ = d * ((2 : ℕ) ^ u1 * r1') := by ring
  -- `d` is odd, so in particular `1 ≤ d`; use this to apply exact division lemma.
  have h1_le_d : 1 ≤ d := by
    exact odd_implies_one_le d hd_odd
  have h_divA : A / d = (2 : ℕ) ^ u1 * r1' := by
    exact nat_div_eq_of_mul_eq h1_le_d hA_mul

  -- 5) rewrite 2 * (A / d) as 2^(u1+1) * r1'
  have h_left : (2 : ℕ) ^ (u1 + 1) * r1' = 2 * (A / d) := by
    calc
      (2 : ℕ) ^ (u1 + 1) * r1' = 2 * ((2 : ℕ) ^ u1 * r1') := by ring
      _ = 2 * (A / d) := by rw [h_divA]

  -- 6) combine with the given decomposition of B to get equality of two 2-adic decompositions
  have h_comb : (2 : ℕ) ^ (u1 + 1) * r1' = (2 : ℕ) ^ u2 * r2 := by
    calc
      (2 : ℕ) ^ (u1 + 1) * r1' = 2 * (A / d) := by exact h_left
      _ = (2 : ℕ) ^ u2 * r2 := by exact hB_eq

  -- 7) show r1' is odd from r1 = d * r1' and oddness of r1 and d
  have hr1'_odd : r1' % 2 = 1 := by
    exact odd_quotient_of_odd_divisor r1 d r1' hr1_eq hr1_odd hd_odd

  -- 8) uniqueness of odd part gives r1' = r2
  have huniq := pow_mul_odd_eq_pow_mul_odd (u1 + 1) u2 r1' r2 h_comb hr1'_odd hr2_odd
  have _ := huniq
  rcases huniq with ⟨_heq_pow, heq_oddpart⟩

  -- 9) conclude r2 = r1' and therefore r2 ≤ r1 because r1 = d * r1' and d ≥ 1
  have heq_r2_r1' : r2 = r1' := by
    exact Eq.symm heq_oddpart
  have hr1_def : r1 = d * r1' := hr1_eq
  have h1_le_d : 1 ≤ d := by
    exact odd_implies_one_le d hd_odd
  have : r2 ≤ r1 := by
    calc
      r2 = r1' := by exact heq_r2_r1'
      _ ≤ d * r1' := by exact mul_le_of_one_le d r1' h1_le_d
      _ = r1 := by rw [hr1_def]
  exact this

/-
Using the decomposition helpers above (left as `sorry`) we now glue them to prove the
nonincrease of the odd part from step k to k+1. The proof below merely arranges the

-/
lemma odd_part_noninc_step
  (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))

    Int.natAbs ((m k : ℤ) - (n k : ℤ)) = (2 : ℕ) ^ u1 * r1 →
    r1 % 2 = 1 → 0 < r1 →
    Int.natAbs ((m (k + 1) : ℤ) - (n (k + 1) : ℤ)) = (2 : ℕ) ^ u2 * r2 →
    r2 % 2 = 1 → 0 < r2 → r2 ≤ r1 := by
  intro k u1 r1 u2 r2 hA hA_odd hA_pos hB hB_odd hB_pos
  -- define the relevant gcd
  let d := Nat.gcd (2 * m k + 1) (2 * n k + 1)

  -- obtain the reduced-form equalities m_{k+1} = (2*m_k+1)/d, n_{k+1} = (2*n_k+1)/d
  have hnext := next_eq_div_gcd m n h_coprime h_recurrence k

  rcases hg with ⟨hd_odd, h_dvd⟩

  -- relate the natAbs of the next difference to natAbs(((2*z)/d))
  have h_step_natabs := diff_evolution_natAbs m n k d hm_succ_eq hn_succ_eq h_dvd

  -- apply the odd-divisor natAbs scaling lemma to rewrite natAbs(((2*z)/d))
  let z : ℤ := (m k : ℤ) - (n k : ℤ)
  have h_scale := natAbs_two_mul_div_of_odd_divisor z d hd_odd h_dvd

  -- combine equalities to get the numeric equality 2 * (A / d) = 2^u2 * r2
  -- where A = Int.natAbs z
  have h_combo : 2 * (Int.natAbs z / d) = (2 : ℕ) ^ u2 * r2 := by
    calc
      2 * (Int.natAbs z / d) = Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) := Eq.symm h_scale
      _ = Int.natAbs ((m (k + 1) : ℤ) - (n (k + 1) : ℤ)) := Eq.symm h_step_natabs

  have hres := odd_part_le_from_div_by_odd (Int.natAbs z) d hdvd_nat u1 r1 u2 r2
    hA hA_odd hA_pos h_combo hB_odd hB_pos hd_odd
  exact hres

  have hdecomp : ∀ k, ∃ u r, Int.natAbs ((m k : ℤ) - (n k : ℤ)) = (2 : ℕ) ^ u * r ∧ r % 2 = 1 ∧ 0 < r := by
    exact decomposition_of_diff m n hΔ

  choose u r hu hr using hdecomp

  -- define the measure t(k) to be the odd part r(k)
  let t : ℕ → ℕ := fun k => r k

  -- positivity: r(k) > 0 by the decomposition property
  have htpos : ∀ k, 0 < t k := by
    intro k
    exact (hr k).2

  -- nonincreasing property: compare odd parts using the decomposition equalities and helper
  have hnoninc : ∀ k, t (k + 1) ≤ t k := by
    intro k
    have Hk := hu k
    have Hk1 := hu (k + 1)

  have hdec : ∀ k, (¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)) → t (k + 1) < t k := by

theorem putnam_2025_a1
  (m n : ℕ → ℕ)
  (hm : ∀ k : ℕ, 0 < m k)
  (hn : ∀ k : ℕ, 0 < n k)
  (h_distinct : m 0 ≠ n 0)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  {k : ℕ | ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)}.Finite := by
  classical

  -- Step 2: differences never vanish
  have hΔ : ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
    exact diff_ne_zero m n h_distinct h_recurrence

  -- Step 4: obtain the decreasing odd-part measure abstractly
  rcases exists_odd_measure m n hΔ h_recurrence h_coprime with ⟨t, ⟨htpos, hnoninc, hdec_step⟩⟩

  -- Step 4/5: define the "bad" property
  let bad : ℕ → Prop := fun k => ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)

  -- Step 5: apply the abstract finiteness lemma using the properties of `t`
  have hfinite : {k : ℕ | bad k}.Finite := by
    exact finite_of_strict_decrease_on_bad t htpos bad hnoninc hdec_step

  -- conclude in the required form
  simpa [bad] using hfinite
