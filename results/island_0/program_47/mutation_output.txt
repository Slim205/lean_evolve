lemma natAbs_two_mul_div_of_odd_divisor (z : ℤ) (d : ℕ)
  (hd_odd : d % 2 = 1) (hdiv : (d : ℤ) ∣ z) :
  Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = 2 * (Int.natAbs z / d) := by
  -- We decompose the divisibility, pass to the integer quotient, and then use multiplicativity
  -- of Int.natAbs together with exact division facts. Glue of the helper lemmas above.
  -- 1) d is nonzero as an integer
  have hd_nz : (d : ℤ) ≠ 0 := by
    exact odd_nat_int_ne_zero d hd_odd

  -- 2) get q with z = d * q
  have hq_exists : ∃ q : ℤ, z = (d : ℤ) * q := by
    exact exists_int_mul_of_dvd_nat d hdiv
  rcases hq_exists with ⟨q, hq⟩

  -- 3) compute ((2*z)/d) = 2*q
  have hdiv_eq : ((2 : ℤ) * z) / (d : ℤ) = (2 : ℤ) * q := by
    exact int_div_mul_eq_two hd_nz hq

  -- 4) take natAbs of both sides and simplify the left using natAbs_two_mul
  have h_left : Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = 2 * Int.natAbs q := by
    calc
      Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = Int.natAbs ((2 : ℤ) * q) := by
        congr
        exact hdiv_eq
      _ = 2 * Int.natAbs q := natAbs_two_mul q

  -- 5) relate Int.natAbs q to Int.natAbs z / d
  have hq_natquot : Int.natAbs z / d = Int.natAbs q := by
    exact natAbs_div_eq_of_mul_eq (hz := hq)

  -- 6) finish by rewriting the right-hand factor
  calc
    Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) = 2 * Int.natAbs q := by exact h_left
    _ = 2 * (Int.natAbs z / d) := by rw [hq_natquot]
