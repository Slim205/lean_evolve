  (h : (2 : ℕ) ^ p * x = (2 : ℕ) ^ q * y) (hx_odd : x % 2 = 1) :
  q ≤ p := by
  -- Proceed by cases: either p < q (impossible because x is odd) or ¬(p < q) which yields q ≤ p.
  by_cases hpq : p < q
  · -- derive a contradiction: from p < q we obtain 2^(p+1) ∣ 2^p * x, hence 2 ∣ x, contradicting oddness
    have hpow : ((2 : ℕ) ^ (p + 1) ∣ ((2 : ℕ) ^ q * y)) := by
      exact pow_succ_dvd_pow_mul p q y hpq
    have hpow' : ((2 : ℕ) ^ (p + 1) ∣ ((2 : ℕ) ^ p * x)) := by
      exact dvd_of_eq h hpow
    have h2divx : 2 ∣ x := by
      exact pow_succ_dvd_pow_mul_implies_two_dvd hpow'
    -- odd_not_two_dvd is defined elsewhere in the file: it yields ¬ (2 ∣ x) from x % 2 = 1
    have hfalse : False := (odd_not_two_dvd hx_odd) h2divx
    exact False.elim hfalse
  · -- otherwise ¬(p < q), hence q ≤ p
    exact Nat.le_of_not_lt hpq
