/-
Using the decomposition helpers above (left as `sorry`) we now glue them to prove the
nonincrease of the odd part from step k to k+1. The proof below merely arranges the
helpers' conclusions and performs trivial transitivity/rewriting steps.

lemma odd_part_noninc_step
  (m n : ℕ → ℕ)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = ((2 : ℚ) * (m k : ℚ) + 1) / ((2 : ℚ) * (n k : ℚ) + 1))
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k)) :
  ∀ k u1 r1 u2 r2,
    Int.natAbs ((m k : ℤ) - (n k : ℤ)) = (2 : ℕ) ^ u1 * r1 →
    r1 % 2 = 1 → 0 < r1 →
    Int.natAbs ((m (k + 1) : ℤ) - (n (k + 1) : ℤ)) = (2 : ℕ) ^ u2 * r2 →
    r2 % 2 = 1 → 0 < r2 → r2 ≤ r1 := by
  intro k u1 r1 u2 r2 hA hA_odd hA_pos hB hB_odd hB_pos
  -- define the relevant gcd
  let d := Nat.gcd (2 * m k + 1) (2 * n k + 1)

  -- obtain the reduced-form equalities m_{k+1} = (2*m_k+1)/d, n_{k+1} = (2*n_k+1)/d
  have hnext := next_eq_div_gcd m n h_coprime h_recurrence k
  rcases hnext with ⟨hm_succ_eq, hn_succ_eq⟩

  -- obtain that d is odd and divides the previous difference
  have hg := gcd_odd_and_dvd_diff m n k
  rcases hg with ⟨hd_odd, h_dvd⟩

  -- relate the natAbs of the next difference to natAbs(((2*z)/d))
  have h_step_natabs := diff_evolution_natAbs m n k d hm_succ_eq hn_succ_eq h_dvd

  -- apply the odd-divisor natAbs scaling lemma to rewrite natAbs(((2*z)/d))
  let z : ℤ := (m k : ℤ) - (n k : ℤ)
  have h_scale := natAbs_two_mul_div_of_odd_divisor z d hd_odd h_dvd

  -- combine equalities to get the numeric equality 2 * (A / d) = 2^u2 * r2
  -- where A = Int.natAbs z
  have h_combo : 2 * (Int.natAbs z / d) = (2 : ℕ) ^ u2 * r2 := by
    calc
      2 * (Int.natAbs z / d) = Int.natAbs (((2 : ℤ) * z) / (d : ℤ)) := Eq.symm h_scale
      _ = Int.natAbs ((m (k + 1) : ℤ) - (n (k + 1) : ℤ)) := Eq.symm h_step_natabs
      _ = (2 : ℕ) ^ u2 * r2 := hB

  -- now invoke the abstract odd-part comparison lemma on A := Int.natAbs z
  have hres := odd_part_le_from_div_by_odd (Int.natAbs z) d u1 r1 u2 r2 hA hA_odd hA_pos h_combo hB_odd hB_pos hd_odd
  exact hres
