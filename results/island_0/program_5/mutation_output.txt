/-- Step 2: the differences never become zero. -/
lemma diff_ne_zero
  (m n : ℕ → ℕ)
  (h_distinct : m 0 ≠ n 0)
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  ∀ k : ℕ, (m k : ℤ) - (n k : ℤ) ≠ 0 := by
  -- instantiate helpers (all are `sorry` above)
  have hden : ∀ k, ((2 : ℚ) * (n k : ℚ) + 1) ≠ 0 ∧ ((2 : ℚ) * (m k : ℚ) + 1) ≠ 0 := by
    exact denom_nonzero_from_recurrence m n h_recurrence
  have hsucc_nonzero : ∀ k, (n (k + 1) : ℚ) ≠ 0 := by
    exact n_succ_nonzero m n h_recurrence
  have hsucc_one := by exact fun k => succ_equal_implies_ratio_one m n hsucc_nonzero k
  have heq_prev := by exact fun k => ratio_one_implies_eq_prev m n (fun k => (hden k).1) k

  -- prove by induction that m k ≠ n k for all k
  have hmn : ∀ k, m k ≠ n k := by
    intro k
    induction k with
    | zero => exact h_distinct
    | succ k ih =>
      by_contra h_eq_succ
      -- if m_{k+1} = n_{k+1} then the left ratio is 1
      have hL : (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = 1 := by
        apply (hsucc_one k)
        exact h_eq_succ
      -- by recurrence the right-hand ratio equals the left, so the right is 1
      have hR : ((2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1) = 1) := by
        calc
          (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1) := h_recurrence k
          _ = 1 := hL
      -- hence m_k = n_k, contradicting IH
      have mk_eq_nk : m k = n k := by
        apply (heq_prev k)
        exact hR
      exact ih mk_eq_nk

  -- finish: translate nat-inequality to integer-subtraction nonzeroness using helper
  intro k
  intro hzero
  have eq_nat : m k = n k := by
    have : (m k : ℤ) - (n k : ℤ) = 0 := hzero
    exact (int_sub_eq_zero_iff_eq_nat (m k) (n k)).1 this
  exact (hmn k) eq_nat
